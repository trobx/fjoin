#' Return the semi-join of the \code{i}-table in a \code{DT[i]}-style data.table
#' join
#'
#' @description Write (and optionally run) \code{data.table} code to return the
#' semi-join of the \code{i}-table in an enhanced \code{DT[i]}-style join.
#' Arguments are as for \code{dtjoin}, except for those controlling the output
#' columns, which do not apply.
#'
#' @inheritParams dtjoin
#' @param nomatch,nomatch.DT For consistency with \code{dtjoin}, these arguments
#'   are permitted, but they have no effect on the resulting semi-join.
#'
#' @returns A \code{data.table} (the resulting semi-join), or \code{NULL} if
#'   \code{do} is \code{FALSE}. The data.table code is always printed to the
#'   console.
#'
#' @examples
#' # TODO
#'
#' @export
dtjoin_semi_i <- function(
    .DT        = NULL,
    .i         = NULL,
    on,
    match.na   = FALSE,
    mult       = "all",
    mult.DT    = "all",
    nomatch    = NULL,
    nomatch.DT = NULL,
    do         = !(is.null(.DT) && is.null(.i)),
    verbose    = FALSE,
    on.first   = FALSE
) {

  check_TF(match.na)
  check_nomatch(nomatch)
  check_nomatch(nomatch.DT)
  check_mult(mult)
  check_mult(mult.DT)
  check_TF(do)
  check_TF(verbose)
  check_TF(on.first)

  check_setup(do, .DT, .i)

  mock        <- is.null(.DT) && is.null(.i)
  has_mult    <- mult != "all"
  has_mult.DT <- mult.DT != "all"

  if (has_mult.DT)
    stop(sprintf("%s doesn't yet implement the argument 'mult.DT'", as.character(match.call()[[1]])))

  # ----------------------------------------------------------------------------

  if (mock) {
    tmp <- make_mock_tables(on)
    .DT <- tmp[[1]]
    .i  <- tmp[[2]]
  }

  # ----------------------------------------------------------------------------

  is_joincol_i <- rep(FALSE, length(names(.i)))

  if (!match.na) {
    equi_names_DT <- character(0)
    equi_names_i  <- character(0)
  }

  for (i in seq_along(on)) {

    s <- strsplit_predicate(on[i])

    idx_DT <- match(s[1], names(.DT))
    if (is.na(idx_DT)) stop(sprintf("No column named \"%s\" found in `.DT`", s[1]))

    idx_i <- match(s[3], names(.i))
    if (is.na(idx_i)) stop(sprintf("No column named \"%s\" found in `.i`", s[3]))

    is_joincol_i[idx_i] <- TRUE

    if (!match.na && s[2] == "==") {
      equi_names_DT <- c(equi_names_DT, s[1])
      equi_names_i  <- c(equi_names_i, s[3])
    }
  }

  # use %in% instead of join if single equality condition (and not mult.DT)
  # TODO will need to adapt for roll
  use_in <- i == 1L && s[2] == "=="

  # we need equi_names_ in this case even if match.na TRUE
  if (use_in && match.na) {
    equi_names_DT <- s[1]
    equi_names_i  <- s[3]
  }

  # TODO if use_in and not on.first, we don't need j at all (and can use setcolorder otherwise)
  # TODO if use_in, we can filter NA's out directly alongside the %in% (may be faster than na.omit then filter)

  jvars <- as.character(names(.i))
  # common cols will get i's values if join cols but otherwise e.g. c=i.c or will get .DT's values
  if (!use_in) jvars <- ifelse(!is_joincol_i & jvars %in% names(.DT), sprintf("%s = i.%s",jvars,jvars), jvars)
  if (on.first) jvars <- c(jvars[is_joincol_i], jvars[!is_joincol_i])
  jtext <- paste0("list(", paste(jvars, collapse = ", "), ")")

  # ----------------------------------------------------------------------------

  # mult "all" to "first" if a join
  if (!use_in) argtext_mult <- sprintf("mult = %s, ", if (!has_mult) "\"first\"" else deparse(mult))
  argtext_verbose <- if (verbose) ", verbose = TRUE" else ""

  screen_NAs <-
    !match.na &&
    length(equi_names_DT) &&
    if (mock) {
      anyNA(.DT[, equi_names_DT]) &&
        anyNA(.i[, equi_names_i])
    } else {
      anyNA(.DT[, equi_names_DT, with = FALSE]) &&
        anyNA(.i[, equi_names_i, with = FALSE])
    }

  .DTtext <- ".DT"
  .itext  <- ".i"
  if (screen_NAs) {
    # we can na.omit on .i if we like as these rows can't be matches
    if (nrow(.DT) < nrow(.i)) {
      .DTtext <- sprintf("na.omit(%s, cols = %s)", .DTtext, deparse(equi_names_DT))
    } else {
      .itext  <- sprintf("na.omit(%s, cols = %s)", .itext, deparse(equi_names_i))
    }
  }

  # ----------------------------------------------------------------------------

  jointext <-
    if (use_in) {
      sprintf("%s[%s %%in%% %s$%s, %s%s]",
              .itext,
              equi_names_i,
              .DTtext,
              equi_names_DT,
              jtext,
              argtext_verbose)
    } else {
      sprintf("%s[%s, on = %s, nomatch = NULL, %s%s%s]",
              .DTtext,
              .itext,
              deparse(on),
              argtext_mult,
              jtext,
              argtext_verbose)
    }

  # ----------------------------------------------------------------------------

  if (!mock) {
    cat(".DT :", deparse(substitute(.DT)), "\n")
    cat(".i  :", deparse(substitute(.i)), "\n")
  }
  cat("Join:", jointext, "\n")

  if (do) eval(parse(text=jointext))
}
